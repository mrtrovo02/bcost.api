'use strict';

import { Module } from '@nestjs/common';
import { EventEmitterModule } from '@nestjs/event-emitter';
import { RevenueService } from './revenue.service.js';
import { RevenueController } from './revenue.controller.js';
import { RevenueRepository } from './repositories/revenue.repository.js';
import { CalculateFactorRUseCase } from './use-cases/calculate-factor-r.use-case.js';
import { CloseMonthUseCase } from './use-cases/close-month.use-case.js';
import { PrismaModule } from '../../database/prisma.module.js';

/**
 * RevenueModule: Gerencia o ciclo de vida financeiro de faturamento.
 * Inclui inteligência fiscal (Fator R) e automação de fechamento.
 */
@Module({
  imports: [
    PrismaModule,
    // Garante que o módulo tenha acesso ao emissor de eventos para logs assíncronos
    EventEmitterModule, 
  ],
  controllers: [RevenueController],
  providers: [
    RevenueService,
    RevenueRepository,
    CalculateFactorRUseCase,
    CloseMonthUseCase,
  ],
  exports: [
    RevenueService, 
    RevenueRepository,
    CalculateFactorRUseCase // Exportado para caso o FiscalModule precise consultar o Fator R
  ],
})
export class RevenueModule {}
