'use strict';

import {
  Controller,
  Post,
  Get,
  Query,
  Param,
  ParseUUIDPipe,
  ParseIntPipe,
  HttpCode,
  HttpStatus,
  Logger,
  UseGuards,
  ForbiddenException,
} from '@nestjs/common';
import {
  ApiTags,
  ApiBearerAuth,
  ApiOperation,
  ApiResponse,
  ApiParam,
  ApiQuery,
  ApiOkResponse,
} from '@nestjs/swagger';

import { RevenueService } from './revenue.service.js';
import { CalculateFactorRUseCase } from './use-cases/calculate-factor-r.use-case.js';
import { JwtAuthGuard } from '../../auth/guards/jwt-auth.guard.js';
import { CompanyAccessGuard } from '../../common/guards/company-access.guard.js';

// DTOs
import { RevenueMetricsResponseDto } from './dto/revenue-metrics-response.dto.js';
import { FactorRResponseDto } from './dto/factor-r-response.dto.js';

@ApiTags('Financial - Revenue (Gestão de Receita)')
@ApiBearerAuth()
@UseGuards(JwtAuthGuard, CompanyAccessGuard) // CompanyAccessGuard garante que o usuário pertence à empresa
@Controller('revenue')
export class RevenueController {
  private readonly logger = new Logger(RevenueController.name);

  constructor(
    private readonly revenueService: RevenueService,
    private readonly calculateFactorRUseCase: CalculateFactorRUseCase,
  ) {}

  @Post('process-billing/:companyId')
  @HttpCode(HttpStatus.OK)
  @ApiOperation({
    summary: 'Processar Faturamento Mensal',
    description: 'Varre contratos ativos e gera faturas para aqueles que vencem hoje via transação atômica.',
  })
  @ApiParam({ name: 'companyId', type: 'string', format: 'uuid' })
  @ApiOkResponse({ description: 'Processamento concluído com sucesso.' })
  async processBilling(
    @Param('companyId', new ParseUUIDPipe()) companyId: string,
  ) {
    this.logger.debug(`🎬 Iniciando processamento de faturamento: ${companyId}`);
    return await this.revenueService.processMonthlyBilling(companyId);
  }

  @Get('metrics/:companyId')
  @ApiOperation({
    summary: 'Métricas de Faturamento e Dashboards',
    description: 'Retorna o total faturado consolidado e o status preditivo do Fator R.',
  })
  @ApiParam({ name: 'companyId', type: 'string', format: 'uuid' })
  @ApiQuery({ name: 'month', required: true, type: Number })
  @ApiQuery({ name: 'year', required: true, type: Number })
  @ApiOkResponse({ type: RevenueMetricsResponseDto })
  async getMetrics(
    @Param('companyId', new ParseUUIDPipe()) companyId: string,
    @Query('month', ParseIntPipe) month: number,
    @Query('year', ParseIntPipe) year: number,
  ): Promise<RevenueMetricsResponseDto> {
    // Validação básica de range de mês
    if (month < 1 || month > 12) throw new ForbiddenException('Mês inválido');

    this.logger.log(`📊 Buscando métricas para empresa ${companyId} - Período: ${month}/${year}`);
    return await this.revenueService.getRevenueMetrics(companyId, month, year);
  }

  @Get('factor-r/:companyId')
  @ApiOperation({
    summary: 'Consulta de Fator R (Deep Analysis)',
    description: 'Analisa faturamento vs folha dos últimos 12 meses conforme legislação do Simples Nacional.',
  })
  @ApiParam({ name: 'companyId', type: 'string', format: 'uuid' })
  @ApiOkResponse({ type: FactorRResponseDto })
  async getFactorR(
    @Param('companyId', new ParseUUIDPipe()) companyId: string,
  ): Promise<FactorRResponseDto> {
    this.logger.debug(`🧠 Executando UseCase de Fator R: ${companyId}`);
    return await this.calculateFactorRUseCase.execute(companyId);
  }
}
